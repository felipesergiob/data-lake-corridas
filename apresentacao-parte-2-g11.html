SELECT uf, sum(coletas) AS coletas,
       round(sum(soma_valor) / sum(coletas), 3) AS preco_medio,
       ...                                          AS diferenca_bandeira
FROM   precos_uf_mes
WHERE  ano = '2025' AND produto = 'GASOLINA'
GROUP BY uf ORDER BY preco_medio DESC LIMIT 10;
